## code

```cpp
class Solution {
public:
    int minimumSumSubarray(vector<int>& nums, int l, int r) {
        int ans{INT32_MAX};

        for (int i = 0; i < nums.size(); ++i) {
            int sum{};
            for (int j = i; j < nums.size(); ++j) {
                sum += nums[j];

                int len = j - i + 1;
                if (len >= l && len <= r && sum > 0) {
                    ans = min(ans, sum);
                }
            }
        }

        return (ans != INT32_MAX) ? ans : -1;
    }
};
```

### complexity

> - Time: `O(n^2)`
> - Space: `O(1)`
